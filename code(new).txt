function doGet() {

  return HtmlService.createHtmlOutputFromFile('Index')

      .setTitle('教科書版本查詢系統')

      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);

}



// 監聽試算表編輯事件：精準捕捉變更的科目、版本與詳細內容

function onEdit(e) {

  if (!e) return;

  const sheet = e.range.getSheet();

  const sheetName = sheet.getName();

  

  if (sheetName !== 'Versions' && sheetName !== 'Students') return;

  

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  let logSheet = ss.getSheetByName('UpdateLog');

  if (!logSheet) {

    logSheet = ss.insertSheet('UpdateLog');

    logSheet.appendRow(['更新時間', '更新內容']);

  }

  

  const row = e.range.getRow();

  const col = e.range.getColumn();

  const newVal = e.value || '無';

  

  let content = `在「${sheetName}」工作表進行了變更`;

  

  try {

    if (sheetName === 'Versions') {

      // 判斷是國小、國中還是高中，並抓取對應欄位

      // 高中從 Q 欄開始 (col 17 起)

      if (col >= 17) {

        let schoolName = sheet.getRange(row, 17).getValue() || ''; // Q欄 學校

        let gradeName = sheet.getRange(row, 18).getValue() || '';  // R欄 年級

        let groupName = sheet.getRange(row, 19).getValue() || '';  // S欄 班群

        

        // 高中科目對應表 (從 T 欄第 20 欄開始分別對應 10 個科目)

        const highSchoolSubjects = ["國文", "英文", "數學", "地理", "歷史", "公民", "生物", "物理", "化學", "地科"];

        let subjectIndex = col - 20; // T 欄是第 20 欄

        

        if (schoolName && subjectIndex >= 0 && subjectIndex < highSchoolSubjects.length) {

          let subjectName = highSchoolSubjects[subjectIndex];

          content = `${schoolName} ${gradeName}${groupName ? ' ' + groupName : ''} ${subjectName} ${newVal}版`;

        } else if (schoolName) {

          content = `更新 ${schoolName} ${gradeName} 版本資料`;

        }

      } else {

        // 國中或國小版本對應 (可根據您的表格結構擴充)

        let schoolName = sheet.getRange(row, 1).getValue() || '';

        let gradeName = sheet.getRange(row, 2).getValue() || '';

        if (schoolName) {

          content = `更新 ${schoolName} ${gradeName} 教科書版本`;

        }

      }

    } else if (sheetName === 'Students') {

      let studentName = sheet.getRange(row, 1).getValue() || '';

      let schoolName = sheet.getRange(row, 2).getValue() || '';

      if (studentName) {

        content = `更新學生資料：${schoolName} - ${studentName}`;

      }

    }

  } catch (err) {

    // 發生例外時保留基本描述

  }



  const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");

  

  logSheet.insertRowAfter(1);

  logSheet.getRange(2, 1, 1, 2).setValues([[now, content]]);

}



// 取得試算表的更新時間與更新內容

function getUpdateLog() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  let logSheet = ss.getSheetByName('UpdateLog');

  

  if (!logSheet) {

    logSheet = ss.insertSheet('UpdateLog');

    logSheet.appendRow(['更新時間', '更新內容']);

    const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");

    logSheet.appendRow([now, "系統初始化完成"]);

  }

  

  const data = logSheet.getDataRange().getValues();

  if (data.length > 1) {

    let latestTime = data[1][0];

    if (latestTime instanceof Date) {

      latestTime = Utilities.formatDate(latestTime, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");

    }

    return {

      time: latestTime ? latestTime.toString() : "未記錄",

      content: data[1][1] ? data[1][1].toString() : "資料已更新"

    };

  }

  

  return {

    time: "尚無記錄",

    content: "尚無更新日誌"

  };

}



// 一次性初始化所有基礎資料，減少連線次數

function getInitialData() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const versionSheet = ss.getSheetByName('Versions');

  

  const options = {

    "國小": ["一年級", "二年級", "三年級", "四年級", "五年級", "六年級"],

    "國中": ["七年級", "八年級", "九年級"],

    "高中": ["10年級", "11年級", "12年級"]

  };



  if (!versionSheet) return { options: options, schools: {}, updateInfo: { time: "", content: "" } };



  const data = versionSheet.getDataRange().getValues();

  data.shift(); 



  let schools = { "國小": [], "國中": [], "高中": [] };

  

  data.forEach(row => {

    if (row[8]) schools["國小"].push(row[8].toString().trim());   

    if (row[0]) schools["國中"].push(row[0].toString().trim());   

    if (row[16]) schools["高中"].push(row[16].toString().trim()); 

  });



  for (let key in schools) {

    schools[key] = [...new Set(schools[key])].filter(v => v !== "").sort();

  }



  const updateInfo = getUpdateLog();



  return {

    options: options,

    schools: schools,

    updateInfo: updateInfo

  };

}



function getFilteredStudents(level, school, grade) {

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Students');

  if (!sheet) return [];

  const data = sheet.getDataRange().getValues();

  data.shift();

  

  return data

    .filter(row => {

      const matchSchool = school ? row[1].toString().trim() === school : true;

      const matchGrade = grade ? row[2].toString().trim() === grade : true;

      return matchSchool && matchGrade;

    })

    .map(row => row[0]);

}



function queryTextbooks(studentName, schoolName, grade, level) {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const versionSheet = ss.getSheetByName('Versions');

  const studentSheet = ss.getSheetByName('Students');

  const versionData = versionSheet.getDataRange().getValues();

  

  let targetSchool = schoolName;

  let targetGrade = grade;

  let targetGroup = "";

  let displayName = studentName ? studentName : "校園版本查詢";



  if (studentName) {

    const sData = studentSheet.getDataRange().getValues();

    let sInfo = sData.find(row => row[0] === studentName);

    if (sInfo) {

      targetSchool = sInfo[1];

      targetGrade = sInfo[2];

      targetGroup = sInfo[3] || "";

    }

  }



  let config = {

    startCol: (level === "國小" ? 8 : (level === "高中" ? 16 : 0)),

    schoolIdx: 0,

    gradeIdx: 1,

    groupIdx: (level === "高中" ? 2 : -1),

    bookStartIdx: (level === "高中" ? 3 : 2),

    subjects: (level === "高中" ? 

      ["國文", "英文", "數學", "地理", "歷史", "公民", "生物", "物理", "化學", "地科"] : 

      ["國文", "英文", "數學", "社會", "自然"])

  };



  let matches = versionData.filter(row => {

    let rowSchool = row[config.startCol + config.schoolIdx];

    let rowGrade = row[config.startCol + config.gradeIdx];

    let matchSchool = rowSchool === targetSchool;

    let matchGrade = targetGrade ? rowGrade === targetGrade : true;

    

    if (studentName && config.groupIdx !== -1 && targetGroup) {

      return matchSchool && matchGrade && row[config.startCol + config.groupIdx] === targetGroup;

    }

    return matchSchool && matchGrade;

  });



  if (matches.length === 0) return `查無 ${targetSchool} 的版本資料`;



  let results = matches.map(match => {

    let books = [];

    config.subjects.forEach((subj, i) => {

      let v = match[config.startCol + config.bookStartIdx + i];

      books.push({ subject: subj, version: (v && v !== "無") ? v : "-" });

    });

    return {

      grade: match[config.startCol + config.gradeIdx],

      group: config.groupIdx !== -1 ? match[config.startCol + config.gradeIdx + 1] : "",

      books: books

    };

  });



  return {

    student: displayName,

    school: targetSchool,

    isMulti: !targetGrade,

    subjects: config.subjects,

    data: results

  };

}

  <body>

    <div class="card">

      <h2>115學年度第一學期教科書版本查詢</h2>

      <div id="updateBanner" class="update-banner">資料載入中...</div>

      

      <div class="selection-container">

        <div class="form-group">

          <label>1. 學制</label>

          <select id="levelSelect" onchange="onLevelChange()">

            <option value="">-- 載入中 --</option>

          </select>

        </div>



        <div class="form-group">

          <label>2. 學校</label>

          <select id="schoolSelect" onchange="onSchoolChange()">

            <option value="">請先選學制</option>

          </select>

        </div>



        <div class="form-group">

          <label>3. 年級 (選填)</label>

          <select id="gradeSelect" onchange="onGradeChange()">

            <option value="">-- 查全年級 --</option>

          </select>

        </div>



        <div class="form-group">

          <label>4. 學生姓名 (選填)</label>

          <select id="studentSelect">

            <option value="">-- 查全校版本 --</option>

          </select>

        </div>

      </div>

      

      <button id="searchBtn" onclick="search()">立即查詢版本</button>

      <div id="result"></div>

    </div>



    <script>

      let cachedData = { options: {}, schools: {}, updateInfo: {} };



      // 初始化：一次抓完所有選項與學校名單

      window.onload = function() {

        google.script.run.withSuccessHandler(res => {

          cachedData = res;

          initLevelSelect();

          displayUpdateInfo(res.updateInfo);

        }).getInitialData();

      };



      function displayUpdateInfo(info) {

        const banner = document.getElementById('updateBanner');

        if (info && info.time) {

          banner.innerHTML = `🕒 最後更新時間：<strong>${info.time}</strong> ｜ 📝 更新內容：${info.content}`;

        } else {

          banner.innerHTML = `🕒 資料已同步完成`;

        }

      }



      function initLevelSelect() {

        const lSelect = document.getElementById('levelSelect');

        lSelect.innerHTML = '<option value="">-- 請選擇 --</option>';

        

        const sortedLevels = ["國小", "國中", "高中"];

        

        sortedLevels.forEach(lvl => {

          if (cachedData.options[lvl]) {

            let opt = document.createElement('option');

            opt.value = opt.textContent = lvl;

            lSelect.appendChild(opt);

          }

        });

      }



      function onLevelChange() {

        const level = document.getElementById('levelSelect').value;

        const sSelect = document.getElementById('schoolSelect');

        const gSelect = document.getElementById('gradeSelect');

        const stSelect = document.getElementById('studentSelect');



        sSelect.innerHTML = '<option value="">-- 選擇學校 --</option>';

        gSelect.innerHTML = '<option value="">-- 查全年級 --</option>';

        stSelect.innerHTML = '<option value="">-- 查全校版本 --</option>';



        if (!level) return;



        const schools = cachedData.schools[level] || [];

        schools.forEach(sch => {

          let opt = document.createElement('option');

          opt.value = opt.textContent = sch;

          sSelect.appendChild(opt);

        });



        if (cachedData.options[level]) {

          cachedData.options[level].forEach(g => {

            let opt = document.createElement('option');

            opt.value = opt.textContent = g;

            gSelect.appendChild(opt);

          });

        }

      }



      function onSchoolChange() { onGradeChange(); }



      function onGradeChange() {

        const level = document.getElementById('levelSelect').value;

        const school = document.getElementById('schoolSelect').value;

        const grade = document.getElementById('gradeSelect').value;

        const stSelect = document.getElementById('studentSelect');

        

        if (school) {

          stSelect.innerHTML = '<option>載入中...</option>';

          google.script.run.withSuccessHandler(list => {

            stSelect.innerHTML = '<option value="">-- 查全校版本 --</option>';

            list.forEach(n => {

              let opt = document.createElement('option');

              opt.value = opt.textContent = n;

              stSelect.appendChild(opt);

            });

          }).getFilteredStudents(level, school, grade);

        }

      }



      function search() {

        const level = document.getElementById('levelSelect').value;

        const school = document.getElementById('schoolSelect').value;

        const grade = document.getElementById('gradeSelect').value;

        const name = document.getElementById('studentSelect').value;

        const resultDiv = document.getElementById('result');

        const btn = document.getElementById('searchBtn');



        if (!level || !school) return alert("請至少選擇學制與學校");

        

        btn.disabled = true;

        btn.innerText = "查詢中...";

        resultDiv.innerHTML = '<p style="text-align:center;">正在抓取最新版本資料，請稍候...</p>';

        

        google.script.run.withSuccessHandler(res => {

          btn.disabled = false;

          btn.innerText = "立即查詢版本";

          

          if (typeof res === 'string') {

            resultDiv.innerHTML = `<p style="color:red; text-align:center;">${res}</p>`;

            return;

          }

          

          let html = `

            <div class="info-box">

              <strong>${res.school}</strong><br>

              <small>查詢對象：${res.student}${res.isMulti ? ' (全校清單)' : ''}</small>

            </div>

            <div class="table-wrapper">

              <table>

                <thead>

                  <tr>

                    <th>年級</th>

                    ${res.data[0].group ? '<th>班群</th>' : ''}

                    ${res.subjects.map(s => `<th>${s}</th>`).join('')}

                  </tr>

                </thead>

                <tbody>

                  ${res.data.map(row => `

                    <tr>

                      <td class="grade-cell">${row.grade}</td>

                      ${row.group ? `<td>${row.group}</td>` : ''}

                      ${row.books.map(b => `<td>${b.version}</td>`).join('')}

                    </tr>

                  `).join('')}

                </tbody>

              </table>

            </div>`;

          resultDiv.innerHTML = html;

        }).queryTextbooks(name, school, grade, level);

      }

    </script>
